/**
 * The conversation the two operators rehearse before a live phone demo.
 *
 * The lines are data. The verdicts are NOT: nothing in this file records
 * what a line is supposed to adjudicate to. /script runs each rep line
 * through the same redact -> stub extraction -> adjudicate path the
 * console uses and prints whatever comes back.
 *
 * That is the point. A crib sheet that asserts "this one goes STALE"
 * is a claim about the system written by hand, which is exactly the kind
 * of thing this product exists to catch. If a line stops producing the
 * verdict we wanted, the page will say so on stage rather than in
 * rehearsal.
 *
 * The customer is Anjali Bhatt: Indore, LG washing machine, Rs 53,700
 * over 9 months. Her EMI figures are nowhere near Meenakshi's Rs 28,400
 * over 6 months, so a misquoted EMI here is visibly about her basket.
 */

  speaker: "rep" | "customer";
  text: string;
  /** Stage direction for the operators. Never shown as dialogue. */

}
