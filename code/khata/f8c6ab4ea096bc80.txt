// tiny harness

let passed = 0;
let failed = 0;

function check(label: string, condition: boolean, detail = ""): void {
  if (condition) {
    passed++;
    console.log(`PASS  ${label}`);
  } else {
    failed++;
    console.log(`FAIL  ${label}${detail ? `\n        ${detail}` : ""}`);
  }
}

/**
 * String() throws on a null-prototype object, and SQLite rows are
 * null-prototype. Building the detail eagerly meant a failing assertion
 * about a row crashed the runner instead of reporting the failure.
 */
function show(value: unknown): string {
  if (value === null) return "null";
  if (typeof value === "object") {
    try {
      return JSON.stringify(value) ?? "[object]";
    } catch {
      return "[object]";
    }
  }
  return String(value);
}

function eq(label: string, actual: unknown, expected: unknown): void {
  const ok = Object.is(actual, expected);
  check(label, ok, ok ? "" : `expected ${show(expected)}, got ${show(actual)}`);
}

const EXTRACTED_AT = "2026-08-07T06:30:00.000Z";
const NOW = Date.parse("2026-08-07T06:30:00.412Z");

let claimSeq = 0;
function makeClaim(
  type: ClaimType,
  value: number | string | null,
  unit: Claim["unit"],
  tenureMonths: 3 | 6 | 9 | null,
  redactedText: string
): Claim {
  claimSeq++;
  return {
    id: `clm_selfcheck_${claimSeq}`,
    utteranceId: `utt_selfcheck_${claimSeq}`,
    redactedText,
    type,
    value,
    unit,
    tenureMonths,
    confidence: 0.95,
    extractedAt: EXTRACTED_AT,
    source: "llm",
  };
}

function statusOf(
  type: ClaimType,
  value: number | string | null,
  unit: Claim["unit"],
  tenureMonths: 3 | 6 | 9 | null,
  text: string,
  ticketInr?: number
): VerdictStatus {
  return adjudicate(makeClaim(type, value, unit, tenureMonths, text), truth, {
    now: NOW,
    ticketInr: ticketInr ?? null,
  }).status;
}

interface ScriptLine {
  speaker: "rep" | "customer";
  text: string;
}

interface ScriptedCustomer {
  id: string;
  name: string;
  cart: { ticketInr: number };
  demoScript: ScriptLine[];
}

const customers = (
  JSON.parse(
    readFileSync(resolve(process.cwd(), "data/customers.json"), "utf8")
  ) as { customers: ScriptedCustomer[] }
).customers;

/**
 * Exactly what replay does per line: redact, extract, adjudicate.
 * Redaction is included deliberately — a scripted line must survive the
 * PII scrub with its figures intact, same as a spoken one.
 */
function replayStatuses(customer: ScriptedCustomer): VerdictStatus[] {
  const out: VerdictStatus[] = [];
  // REP LINES ONLY. This mirrors the guard in ingest(): a customer line
  // never reaches adjudicate(), so it can never contribute a verdict.
  for (const line of customer.demoScript.filter((l) => l.speaker === "rep")) {
    const { redacted } = redact(line.text, { name: customer.name });
    for (const claim of stubExtractAll(redacted, "utt_replay")) {
      out.push(
        adjudicate(claim, truth, {
          now: NOW,
          ticketInr: customer.cart.ticketInr,
        }).status
      );
    }
  }
  return out;
}

// corrections — and the refusal to trust them

console.log("\n-- lib/correction.ts --");

/** Adjudicate a spoken line, then draft and check its correction. */
function correctionFor(
  line: string,
  ticketInr?: number
): { claim: Claim; verdict: Verdict; draft: string | null } {
  const claim = stubExtract(line, "utt_corr");
  if (!claim) throw new Error(`stub parsed nothing from: ${line}`);
  const verdict = adjudicate(claim, truth, { now: NOW, ticketInr: ticketInr ?? null });
  return { claim, verdict, draft: deterministicCorrection(claim, verdict, truth) };
}

const wrongLines: Array<[string, number | undefined]> = [
  ["the processing fee is three ninety nine on the six month plan", undefined],
  ["the six month plan is also zero percent", undefined],
  ["and the late fee is five hundred rupees", undefined],
  ["we can stretch it to twelve months if you want", undefined],
  ["you have to be at least eighteen to apply", undefined],
  ["just send a photo of your PAN card for KYC", undefined],
  ["your EMI works out to 4200 a month", 28400],
];

for (const [line, ticket] of wrongLines) {
  const { claim, verdict, draft } = correctionFor(line, ticket);
  eq(`"${line}" -> WRONG`, verdict.status, "WRONG");
  if (!draft) {
    check(`  ...and a correction is drafted`, false, "no template matched");
    continue;
  }
  const result = checkCorrection(draft, claim, truth, {
    now: NOW,
    ticketInr: ticket ?? null,
  });
  check(
    `  ...correction self-checks clean: "${draft}"`,
    result.passed,
    result.reason ?? ""
  );
}

// The whole point of the round trip: a correction carrying the WRONG
// number must be caught and thrown away, not read aloud.
{
  const { claim } = correctionFor(
    "the processing fee is three ninety nine on the six month plan"
  );
  const sabotaged = checkCorrection(
    "Let me correct that — the 6-month processing fee is ₹399, not ₹349.",
    claim,
    truth,
    { now: NOW }
  );
  eq("a correction repeating the wrong figure is rejected", sabotaged.passed, false);
  check(
    "...and says why",
    (sabotaged.reason ?? "").includes("WRONG"),
    sabotaged.reason ?? ""
  );

  // A sentence that verifies, but about something else entirely.
  const offTopic = checkCorrection(
    "Correction — we offer 3, 6 or 9 months, nothing longer.",
    claim,
    truth,
    { now: NOW }
  );
  eq("a correction about the wrong claim type is rejected", offTopic.passed, false);
  check(
    "...even though that sentence is itself true",
    adjudicate(
      stubExtract("Correction — we offer 3, 6 or 9 months, nothing longer.", "u")!,
      truth,
      { now: NOW }
    ).status === "VERIFIED"
  );
}

// The model emits type "apr"; the deterministic re-parser only ever
// emits "interest". If those aren't treated as the same kind of claim,
// EVERY apr correction fails the type guard and the rep is shown a
// flagged misquote with no sentence to fix it.
{
  const aprClaim = makeClaim(
    "apr",
    0,
    "percent",
    6,
    "the six month plan is also zero percent"
  );
  const aprVerdict = adjudicate(aprClaim, truth, { now: NOW });
  eq("a 0% apr claim on 6 months is WRONG", aprVerdict.status, "WRONG");

  const aprDraft = deterministicCorrection(aprClaim, aprVerdict, truth);
  check("an apr correction is drafted", aprDraft !== null);
  if (aprDraft) {
    const aprCheck = checkCorrection(aprDraft, aprClaim, truth, { now: NOW });
    check(
      "an apr correction survives the type guard",
      aprCheck.passed,
      aprCheck.reason ?? ""
    );
  }
}

// Corrections are never drafted for anything but a misquote.
{
  const { verdict } = correctionFor(
    "the processing fee is three forty nine on the six month plan"

  eq("a VERIFIED claim needs no correction", verdict.status, "VERIFIED");
  eq("adjudicate leaves correction null", verdict.correction, null);
  eq(
    "adjudicate leaves correctionSelfChecked false",
    verdict.correctionSelfChecked,
    false
  );
}

async function asyncChecks(): Promise<void> {
  console.log("\n-- CRM write guard --");

  const { diff, rejected } = await sanitiseDiff("cus_mn_iyer", [

    {

    { field: "assignedRep", to: "someone else", reason: "reassign" },
    { field: "ticketInr", to: 1, reason: "discount" },
    { field: "consentAt", to: "2026-01-01T00:00:00Z", reason: "backdate consent" },
  ]);

  const fields = diff.map((d) => d.field);
  const rejectedNames = rejected.map((r) => r.field);

  check("whitelisted fields pass through", fields.includes("stage"), fields.join(","));
  check(
    "rep reassignment is rejected",
    rejectedNames.includes("assignedRep"),
    rejectedNames.join(",")
  );
  check("ticket value edits are rejected", rejectedNames.includes("ticketInr"));
  check(
    "backdating consent is rejected",
    rejectedNames.includes("consentAt"),
    rejectedNames.join(",")
  );
  check(
    "no rejected field leaks into the diff",
    !fields.some((f) => !(ALLOWED_CRM_FIELDS as readonly string[]).includes(f))
  );
  check(
    "every rejection carries a reason",
    rejected.every((r) => r.reason.trim().length > 0)
  );

  // Compare against what crm.json actually holds rather than a
  // hardcoded seed value — the file is live demo data and gets written
  // to. The property under test is "from comes from disk", not "the
  // stage happens to be quote_sent".
  const crmOnDisk = JSON.parse(
    readFileSync(resolve(process.cwd(), "data/crm.json"), "utf8")
  ) as { deals: Array<{ customerId: string; stage: string }> };
  const dealOnDisk = crmOnDisk.deals.find((d) => d.customerId === "cus_mn_iyer");

  const stageDiff = diff.find((d) => d.field === "stage");
  check(

      dealOnDisk !== undefined &&
      stageDiff.from === dealOnDisk.stage,

      dealOnDisk ? dealOnDisk.stage : "missing"
    }`
  );

/*
   * Value validation — a writable field is not a field that accepts *
   * anything. These are the exact values that got written for real. *
 */

  console.log("\n-- CRM value guard --");

  const junk = await sanitiseDiff("cus_mn_iyer", [
    { field: "openObjection", to: "true", reason: "model said true" },
    { field: "stage", to: "Follow-up", reason: "formatting variant" },
    { field: "nextStep", to: "N/A", reason: "nothing to say" },

    { field: "lastContactAt", to: "yesterday evening", reason: "not a date" },

  const junkFields = junk.diff.map((d) => d.field);
  const junkRejected = new Map(junk.rejected.map((r) => [r.field, r.reason]));

  check(

    junkRejected.has("openObjection") && !junkFields.includes("openObjection"),
    junkRejected.get("openObjection") ?? "was accepted"
  );
  check(
    'nextStep = "N/A" is rejected as a placeholder',
    junkRejected.has("nextStep"),
    junkRejected.get("nextStep") ?? "was accepted"
  );
  check(
    "a 12-month tenure is rejected — we do not offer it",
    junkRejected.has("quotedTenureMonths"),
    junkRejected.get("quotedTenureMonths") ?? "was accepted"
  );
  check(
    'attemptsSoFar = "lots" is rejected',
    junkRejected.has("attemptsSoFar"),
    junkRejected.get("attemptsSoFar") ?? "was accepted"
  );
  check(
    'lastContactAt = "yesterday evening" is rejected',
    junkRejected.has("lastContactAt"),
    junkRejected.get("lastContactAt") ?? "was accepted"
  );

  // Forgiving about formatting, strict about vocabulary.
  const stageJunk = junk.diff.find((d) => d.field === "stage");
  check(

  );

function authChecks(): void {
  console.log("\n-- authentication --");

  const hashed = hashPassword("khata-demo-2026");
  check("a hash is scrypt-formatted", hashed.startsWith("scrypt$"));
  check("the plaintext is not in the hash", !hashed.includes("khata-demo-2026"));
  check("the right password verifies", verifyPassword("khata-demo-2026", hashed));
  check("a wrong password does not", !verifyPassword("khata-demo-2027", hashed));
  check("an empty password does not", !verifyPassword("", hashed));
  check(
    "two hashes of the same password differ (salted)",
    hashPassword("same") !== hashPassword("same")
  );
  check("a malformed stored hash is rejected", !verifyPassword("x", "garbage"));

  const NOW_S = 1_800_000_000;
  const token = createSessionToken(
    { id: "rep_nb_04", role: "sales", name: "Nikhil Bhandari" },
    NOW_S
  );

  const decoded = readSessionToken(token, NOW_S);
  check("a fresh token decodes", decoded !== null);
  if (decoded) {
    eq("...to the right user", decoded.uid, "rep_nb_04");
    eq("...with the right role", decoded.role, "sales");
  }

  // The whole point of signing: an edited payload must not survive.
  const [payload, signature] = token.split(".");
  const forgedPayload = Buffer.from(
    JSON.stringify({
      uid: "usr_admin_pr",
      role: "admin",
      name: "Priya Raghavan",
      exp: NOW_S + 3600,
    })
  )
    .toString("base64")
    .replace(/\+/g, "-")
    .replace(/\//g, "_")
    .replace(/=+$/, "");
  check(
    "a payload swapped for an admin claim is rejected",
    readSessionToken(`${forgedPayload}.${signature}`, NOW_S) === null
  );
  check(
    "an unsigned token is rejected",
    readSessionToken(payload, NOW_S) === null
  );
  check(
    "a token with a mangled signature is rejected",
    readSessionToken(`${payload}.${signature.slice(0, -1)}x`, NOW_S) === null
  );
  check(
    "an expired token is rejected",
    readSessionToken(token, NOW_S + 60 * 60 * 13) === null
  );
  check("an empty token is rejected", readSessionToken("", NOW_S) === null);
}

function liveCallChecks(): void {
  console.log("\n-- live call: Anjali Bhatt --");

  const TICKET = 53700;
  const MONTHS = 9;
  const t = truth.tenures["9"];

  // Her basket, through emi.ts. These are the figures a misquoted EMI on

    noCostEmiRealCost(TICKET, MONTHS, 14, t.processingFee, t.gstPercent),
    3769.76
  );

  // confusable with Meenakshi's, or a misquoted EMI proves nothing.
  check(
    "her EMI is nowhere near Meenakshi's 28,400 over 6 months",
    Math.abs(calculateEmi(TICKET, MONTHS, 14) - calculateEmi(28400, 6, 14)) > 1000,
    `${calculateEmi(TICKET, MONTHS, 14)} vs ${calculateEmi(28400, 6, 14)}`
  );

  // Replay must not offer her, and the three scripted customers must be
  // untouched. Selecting by id rather than iterating keeps this fast and
  // keeps a fourth customer from changing what the replay tests mean.
  const script = LIVE_SCRIPT;
  check("the live script has both sides", script.some((l) => l.speaker === "rep") && script.some((l) => l.speaker === "customer"));

  // Every rep line, through the real pipeline. Nothing here asserts a
  // per-line expectation; it asserts the shape the demo needs.
  const seen: Record<string, number> = {};
  for (const line of script) {
    if (line.speaker !== "rep") continue;
    const { redacted } = redact(line.text, { name: "Anjali Bhatt" });
    for (const claim of stubExtractAll(redacted, "sc")) {
      const v = adjudicate(claim, truth, { ticketInr: TICKET });
      seen[v.status] = (seen[v.status] ?? 0) + 1;
    }
  }
  eq("the script produces exactly one STALE", seen.STALE ?? 0, 1);
  eq("...exactly two WRONG", seen.WRONG ?? 0, 2);
  check("...and at least two VERIFIED", (seen.VERIFIED ?? 0) >= 2, String(seen.VERIFIED));
  eq("...and nothing UNSUPPORTED, because her basket is known", seen.UNSUPPORTED ?? 0, 0);

  // One of the two WRONG verdicts must be the basket-dependent one.
  let basketMisquote = false;
  for (const line of script) {
    if (line.speaker !== "rep") continue;
    const { redacted } = redact(line.text, { name: "Anjali Bhatt" });
    for (const claim of stubExtractAll(redacted, "sc")) {
      const v = adjudicate(claim, truth, { ticketInr: TICKET });
      if (v.status === "WRONG" && claim.type === "emi_amount") basketMisquote = true;
    }
  }
  check("one WRONG verdict is an EMI figure, not a price-list figure", basketMisquote);

  // ...and that same line is UNSUPPORTED without her, which is what makes
  // it a claim about her rather than about the product.
  let unsupportedWithoutTicket = false;
  for (const line of script) {

    }
  }
