/**
 * The header every authenticated page wears.
 *
 * The link list comes from navLinks(role) in lib/access.ts and is not
 * re-derived anywhere. That is the whole point: a role cannot see a link
 * it cannot open, because the list is a function of the role rather than
 * a fixed list with things conditionally hidden.
 *
 * `current` marks the page you are on. It is not a link, it is not
 * underlined, and it carries a rule beneath it — on a projector the
 * underline-versus-no-underline distinction alone does not survive.
 */

import Link from "next/link";

import { LogoutButton } from "@/app/logout-button";
import { navLinks } from "@/lib/access";
import type { Role } from "@/lib/access";

export function AppHeader({
  role,
  name,
  page,
  current,
  authOff = false,
}: {
  role: Role;
  name: string;
  /** What this page is, in lowercase. Sits beside the wordmark. */
  page: string;
  /** Path of the page being rendered, or "" for a detail route. */
  current: string;
  authOff?: boolean;
}) {
  return (
    <header className="border-b border-ink/30 pb-3">
      <div className="flex flex-wrap items-baseline gap-x-4 gap-y-2">
        <Link
          href="/dashboard"
          className="font-sans text-2xl font-semibold tracking-tight text-ink focus-visible:outline-2 focus-visible:outline-offset-4 focus-visible:outline-ink"
        >
          Khata
        </Link>
        <p className="font-sans text-sm text-ink/60">{page}</p>

        {authOff && (
          <span className="border border-[#A9761B] px-2 py-0.5 font-mono text-[10px] uppercase tracking-[0.14em] text-[#A9761B]">
            auth off
          </span>
        )}

        <div className="ml-auto flex items-center gap-4">
          <span className="font-mono text-xs text-ink/60">{name}</span>
          <span className="border border-hairline px-2 py-0.5 font-mono text-[10px] uppercase tracking-[0.14em] text-ink/60">
            {role}
          </span>
          <LogoutButton />
        </div>
      </div>

      <nav className="mt-3 flex flex-wrap items-baseline gap-x-7 gap-y-1">
        {navLinks(role).map((link) =>
          link.href === current ? (
            <span
              key={link.href}
              aria-current="page"
              className="border-b-2 border-ink pb-1 font-sans text-sm font-medium text-ink"
            >
              {link.label}
            </span>
          ) : (
            <Link
              key={link.href}
              href={link.href}
              className="border-b-2 border-transparent pb-1 font-sans text-sm text-ink/55 transition-colors hover:border-hairline hover:text-ink focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-ink"
            >
              {link.label}
            </Link>
          )
        )}
      </nav>
    </header>
  );
}
