PRAGMA journal_mode = WAL;
PRAGMA foreign_keys = ON;

CREATE TABLE IF NOT EXISTS users (
  id            TEXT PRIMARY KEY,
  name          TEXT NOT NULL,
  email         TEXT NOT NULL UNIQUE,
  password_hash TEXT NOT NULL,
  role          TEXT NOT NULL CHECK (role IN ('admin','sales')),
  created_at    TEXT NOT NULL
);

CREATE TABLE IF NOT EXISTS customers (
  id               TEXT PRIMARY KEY,
  name             TEXT NOT NULL,
  city             TEXT NOT NULL,
  product          TEXT NOT NULL,
  ticket_inr       REAL NOT NULL,
  tenure_months    INTEGER NOT NULL CHECK (tenure_months IN (3,6,9)),
  phone            TEXT,
  demo_script      TEXT NOT NULL,          -- JSON array, content not relational
  assigned_user_id TEXT REFERENCES users(id)
);

CREATE TABLE IF NOT EXISTS deals (
  customer_id         TEXT PRIMARY KEY REFERENCES customers(id),
  stage               TEXT NOT NULL,
  open_objection      TEXT,
  next_step           TEXT,
  quoted_tenure_months INTEGER,
  attempts_so_far     INTEGER NOT NULL DEFAULT 0,
  last_contact_at     TEXT,
  approved_by_user_id  TEXT REFERENCES users(id),
  last_call_id         TEXT
);

CREATE TABLE IF NOT EXISTS calls (
  call_id           TEXT PRIMARY KEY,
  user_id           TEXT NOT NULL REFERENCES users(id),
  customer_id       TEXT NOT NULL REFERENCES customers(id),
  started_at        TEXT NOT NULL,
  ended_at          TEXT,
  consent_at        TEXT,
  claims_checked    INTEGER NOT NULL DEFAULT 0,
  verified_count    INTEGER NOT NULL DEFAULT 0,
  wrong_count       INTEGER NOT NULL DEFAULT 0,
  stale_count       INTEGER NOT NULL DEFAULT 0,
  unsupported_count INTEGER NOT NULL DEFAULT 0,
  coverage_complete INTEGER NOT NULL DEFAULT 0,
  cost_inr          REAL NOT NULL DEFAULT 0
);

CREATE TABLE IF NOT EXISTS khatas (
  call_id    TEXT PRIMARY KEY REFERENCES calls(call_id),
  hash       TEXT NOT NULL,
  prev_hash  TEXT,
  algorithm  TEXT NOT NULL,
  iv         TEXT NOT NULL,
  auth_tag   TEXT NOT NULL,
  ciphertext TEXT NOT NULL,
  created_at TEXT NOT NULL
);

CREATE INDEX IF NOT EXISTS idx_calls_user     ON calls(user_id);
CREATE INDEX IF NOT EXISTS idx_calls_customer ON calls(customer_id);
CREATE INDEX IF NOT EXISTS idx_cust_assigned  ON customers(assigned_user_id);
CREATE INDEX IF NOT EXISTS idx_deals_approver ON deals(approved_by_user_id);
