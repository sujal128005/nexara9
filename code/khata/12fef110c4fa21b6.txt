/**
 * Role-aware dashboard at /dashboard.
 *
 * One route, two views, chosen by the session role on the server. There
 * is no ?view= parameter and no client-side branch — a sales user cannot
 * reach the admin view because the admin data is never fetched for them,
 * not because a component was hidden.
 *
 * Every number comes from lib/metrics.ts over khatas that were actually
 * written. Where a metric has no denominator it prints "insufficient
 * data" rather than a zero.
 */

import Link from "next/link";

import { AppHeader } from "@/app/header";
import {
  TABLE,
  TD,
  TD_LINK,
  TD_META,
  TD_NUM,
  TD_NUM_MUTED,
  TH,
  TH_NUM,
  THEAD_ROW,
  TR,
} from "@/app/table";
import { ADMIN_READ_ONLY_NOTE, CONSOLE_DENIED_MESSAGE } from "@/lib/access";
import { listKhatas, readApproval } from "@/lib/closer";
import {
  isInitialised,
  listCallsForUser,
  listCustomers,
  listUsers,
} from "@/lib/db";
import {
  byRep,
  factsFromKhata,
  fieldLabel,
  misquotedFields,
  topMisquotedField,
  totals,
} from "@/lib/metrics";
import type { CallFacts } from "@/lib/metrics";
import { currentSession } from "@/lib/session";

export const dynamic = "force-dynamic";

const NO_DATA = "insufficient data";
const LABEL = "font-mono text-[11px] uppercase tracking-[0.14em] text-ink/40";

function rupees(n: number): string {
  return "₹" + n.toFixed(2);
}

function percent(rate: number | null): string {
  return rate === null ? NO_DATA : `${(rate * 100).toFixed(1)}%`;
}

function shortDate(iso: string): string {
  const d = new Date(iso);
  if (Number.isNaN(d.getTime())) return "—";
  const pad = (n: number) => String(n).padStart(2, "0");
  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())} ${pad(
    d.getHours()
  )}:${pad(d.getMinutes())}`;
}

function Shell({ children }: { children: React.ReactNode }) {
  return (
    <main className="mx-auto flex min-h-screen max-w-6xl flex-col gap-12 bg-paper px-8 py-12">
      {children}
    </main>
  );
}

function Stat({
  label,
  value,
  note,
}: {
  label: string;
  value: string;
  note?: string;
}) {
  return (
    <div className="border-t border-hairline pt-3">
      <span className={LABEL}>{label}</span>
      <p
        className={`mt-1.5 font-mono text-lg ${
          value === NO_DATA ? "text-ink/35" : "text-ink"
        }`}
      >
        {value}
      </p>
      {note && <p className="font-mono text-[11px] text-ink/40">{note}</p>}
    </div>
  );
}

export default async function DashboardPage({
  searchParams,
}: {
  searchParams: Promise<{ denied?: string }>;
}) {
  const { denied } = await searchParams;

  if (!isInitialised()) {
    return (
      <Shell>
        <p className="max-w-2xl font-sans text-sm leading-relaxed text-ink/70">
          The database has not been initialised. Run{" "}
          <span className="font-mono text-xs">npm run db:init</span> and then{" "}
          <span className="font-mono text-xs">npm run reset:demo</span>.
        </p>
      </Shell>
    );
  }

  const session = await currentSession();
  if (!session) {
    return (
      <Shell>
        <p className="font-sans text-sm text-ink/70">Not signed in.</p>
      </Shell>
    );
  }

  const isAdmin = session.role === "admin";

  // The scope. listCallsForUser returns every call for an admin and only
  // their own for a sales user, as a WHERE clause — so a rep's khata is
  // never even decrypted for someone who should not see it.
  const visibleCalls = listCallsForUser(session.uid, session.role);
  const repByCall = new Map(visibleCalls.map((c) => [c.call_id, c.user_id]));
  const visibleIds = new Set(repByCall.keys());

  const khatas = (await listKhatas()).filter((k) => visibleIds.has(k.callId));

  const facts: CallFacts[] = [];
  for (const khata of khatas) {
    const approval = await readApproval(khata.callId);
    facts.push(
      factsFromKhata(khata, repByCall.get(khata.callId) ?? "", approval !== null)
    );
  }
  facts.sort((a, b) => (a.endedAt < b.endedAt ? 1 : -1));

  const overall = totals(facts);
  const topField = topMisquotedField(facts);

  const names = new Map<string, string>();
  for (const c of listCustomers()) names.set(c.id, c.name);
  for (const u of listUsers()) names.set(u.id, u.name);

  const header = (
    <>
      <AppHeader
        role={session.role}
        name={session.name}
        page={isAdmin ? "team dashboard" : "my dashboard"}
        current="/dashboard"
      />
      <section className="-mt-4">
        <h1 className="font-sans text-xl font-semibold tracking-tight text-ink">
          {isAdmin ? "Team record" : "My record"}
        </h1>
        <p className="mt-1 max-w-2xl font-sans text-sm leading-relaxed text-ink/60">
          {isAdmin
            ? "Every closed call, aggregated by rep, so you can see which product field the team gets wrong most often."
            : "Your closed calls, and the product field you misquote most often."}
        </p>
      </section>
    </>
  );

  // Shown when an admin was bounced off /console. It says why, because a
  // redirect with no explanation reads as a bug.
  const denial =
    denied === "console" && isAdmin ? (
      <p className="max-w-3xl border-l-2 border-ink pl-4 font-sans text-sm leading-relaxed text-ink">
        {CONSOLE_DENIED_MESSAGE}
      </p>
    ) : null;

  const readOnlyNote = isAdmin ? (
    <p className="max-w-3xl font-sans text-sm leading-relaxed text-ink/60">
      {ADMIN_READ_ONLY_NOTE}
    </p>
  ) : null;

  if (facts.length === 0) {
    return (
      <Shell>
        {header}
        {denial}
        {readOnlyNote}
        <section className="max-w-2xl">
          <h2 className="font-sans text-lg font-medium text-ink">
            {isAdmin
              ? "Nobody has closed a call yet"
              : "You have not closed a call yet"}
          </h2>
          <p className="mt-2 font-sans text-sm leading-relaxed text-ink/70">
            Once a call is closed this page will show{" "}
            {isAdmin ? "the team's" : "your"} misquote rate, the product field{" "}
            {isAdmin ? "the team gets" : "you get"} wrong most often, how much
            of each call was actually checked, and what the model cost.
          </p>
          <p className="mt-3 font-sans text-sm leading-relaxed text-ink/60">
            The data comes from khatas, and a khata is written when a call is
            closed in the console — either from the microphone or from replay
            mode, which runs a scripted call through the same pipeline. Nothing
            on this page is seeded, so an empty page here means no calls, not a
            broken query.
          </p>
        </section>
      </Shell>
    );
  }

  /* admin */

  if (isAdmin) {
    const reps = byRep(facts);
    const allFields = misquotedFields(facts);

    return (
      <Shell>
        {header}
        {denial}
        {readOnlyNote}

        {/* The coaching signal, given the most space on the page. A
            misquote rate tells you there is a problem; this tells you
            what to run the next team session about. */}
        <section className="border-b border-hairline pb-8">
          <h2 className={LABEL}>The field the team gets wrong most</h2>
          {topField === null ? (
            <>
              <p className="mt-3 font-mono text-2xl text-ink/35">{NO_DATA}</p>
              <p className="mt-2 max-w-2xl font-sans text-sm leading-relaxed text-ink/60">
                No misquotes recorded across {overall.calls} call
                {overall.calls === 1 ? "" : "s"}. There is nothing to coach yet.
              </p>
            </>
          ) : (
            <>
              <p className="mt-3 font-sans text-3xl font-semibold tracking-tight text-ink">
                {fieldLabel(topField.field)}
              </p>
              <p className="mt-2 font-mono text-xs text-ink/50">
                {topField.field} &nbsp;·&nbsp; wrong {topField.count} time
                {topField.count === 1 ? "" : "s"} of {overall.misquotes} total
                misquote{overall.misquotes === 1 ? "" : "s"}
              </p>
              {allFields.length > 1 && (
                <ul className="mt-5 flex flex-wrap gap-x-8 gap-y-2">
                  {allFields.slice(1).map((f) => (
                    <li key={f.field} className="font-mono text-[11px] text-ink/50">
                      {fieldLabel(f.field)}
                      <span className="ml-2 text-ink/35">{f.count}</span>
                    </li>
                  ))}
                </ul>
              )}
            </>
          )}
        </section>

        <section>
          <h2 className={LABEL}>Team totals</h2>
          <div className="mt-4 grid grid-cols-2 gap-x-10 gap-y-7 md:grid-cols-4">
            <Stat label="Calls" value={String(overall.calls)} />
            <Stat label="Claims checked" value={String(overall.claims)} />
            <Stat
              label="Misquote rate"
              value={percent(overall.misquoteRate)}
              note={
                overall.misquoteRate === null
                  ? undefined
                  : `${overall.misquotes} of ${overall.verdicts} claims`
              }
            />
            <Stat
              label="Fully checked calls"
              value={`${overall.fullyCheckedCalls}/${overall.calls}`}
              note={
                overall.fullyCheckedCalls === overall.calls
                  ? "every call fully verified"
                  : "the rest are partially verified"
              }
            />
            <Stat label="Total model cost" value={rupees(overall.costInr)} />
            <Stat
              label="Cost per claim"
              value={
                overall.costPerClaim === null
                  ? NO_DATA
                  : rupees(overall.costPerClaim)
              }
            />
            <Stat
              label="Corrections offered"
              value={String(overall.correctionsOffered)}
              note="self-checked before display"
            />
            <Stat
              label="Proposals approved"
              value={`${overall.approvedCalls}/${overall.calls}`}
            />
          </div>
        </section>

        <section>
          <h2 className={LABEL}>By rep — worst misquote rate first</h2>
          <table className={TABLE}>
            <thead>
              <tr className={THEAD_ROW}>
                <th className={TH}>Rep</th>
                {[
                  "Calls",
                  "Claims",
                  "Wrong",
                  "Misquote rate",
                  "Fully checked",
                  "Cost",
                ].map((label) => (
                  <th key={label} className={TH_NUM}>
                    {label}
                  </th>
                ))}
              </tr>
            </thead>
            <tbody>
              {reps.map((rep) => (
                <tr key={rep.userId} className={TR}>
                  <td className={TD}>
                    <Link href={`/reps/${rep.userId}`} className={TD_LINK}>
                      {names.get(rep.userId) ?? rep.userId}
                    </Link>
                  </td>
                  <td className={TD_NUM_MUTED}>{rep.calls}</td>
                  <td className={TD_NUM_MUTED}>{rep.claims}</td>
                  <td className={TD_NUM}>{rep.misquotes}</td>
                  <td
                    className={
                      rep.misquoteRate === null
                        ? `${TD_NUM} text-ink/35`
                        : TD_NUM
                    }
                  >
                    {percent(rep.misquoteRate)}
                  </td>
                  <td className={TD_NUM_MUTED}>
                    {rep.fullyCheckedCalls}/{rep.calls}
                  </td>
                  <td className={TD_NUM_MUTED}>{rupees(rep.costInr)}</td>
                </tr>
              ))}
            </tbody>
          </table>
          <p className="mt-3 max-w-3xl font-mono text-[11px] leading-relaxed text-ink/40">
            A rep with no checked claims shows{" "}
            <span className="text-ink/60">{NO_DATA}</span> and sorts last. An
            empty rate is not a good rate.
          </p>
        </section>
      </Shell>
    );
  }

  /* sales */

  return (
    <Shell>
      {header}

      <section>
        <h2 className={LABEL}>My calls</h2>
        <div className="mt-4 grid grid-cols-2 gap-x-10 gap-y-7 md:grid-cols-4">
          <Stat label="Calls made" value={String(overall.calls)} />
          <Stat label="Claims checked" value={String(overall.claims)} />
          <Stat
            label="My misquote rate"
            value={percent(overall.misquoteRate)}
            note={
              overall.misquoteRate === null
                ? undefined
                : `${overall.misquotes} of ${overall.verdicts} claims`
            }
          />
          <Stat
            label="Fully checked"
            value={`${overall.fullyCheckedCalls}/${overall.calls}`}
          />
        </div>
      </section>

      <section className="border-t border-hairline pt-8">
        <h2 className={LABEL}>What I get wrong most</h2>
        {topField === null ? (
          <p className="mt-3 font-mono text-lg text-ink/35">{NO_DATA}</p>
        ) : (
          <>
            <p className="mt-3 font-sans text-2xl font-semibold tracking-tight text-ink">
              {fieldLabel(topField.field)}
            </p>
            <p className="mt-2 font-mono text-xs text-ink/50">
              {topField.field} &nbsp;·&nbsp; {topField.count} time
              {topField.count === 1 ? "" : "s"}
            </p>
          </>
        )}
      </section>

      <section className="border-t border-hairline pt-8">
        <h2 className={LABEL}>Corrections</h2>
        <div className="mt-4 grid grid-cols-2 gap-x-10 gap-y-7 md:grid-cols-4">
          <Stat
            label="Offered"
            value={String(overall.correctionsOffered)}
            note="shown, after passing the self-check"
          />
          <Stat label="Accepted" value={NO_DATA} note="not measured — see below" />
        </div>
        <p className="mt-4 max-w-2xl font-sans text-sm leading-relaxed text-ink/60">
          Khata can tell you a correction was put on screen. It cannot tell you
          whether you read it out, because nothing in the pipeline observes
          that. Rather than substitute a proxy — approvals, or call outcomes —
          and let it be read as an acceptance rate, this is left unmeasured.
          Measuring it properly needs one more signal: a control in the console
          for marking a correction as said.
        </p>
      </section>

      <section className="border-t border-hairline pt-8">
        <h2 className={LABEL}>Recent calls</h2>
        <table className={TABLE}>
          <thead>
            <tr className={THEAD_ROW}>
              <th className={TH}>Customer</th>
              <th className={TH}>Ended</th>
              {[
                "Claims",
                "Verified",
                "Wrong",
                "Stale",
                "Unsupported",
                "Coverage",
                "Cost",
              ].map((label) => (
                <th key={label} className={TH_NUM}>
                  {label}
                </th>
              ))}
            </tr>
          </thead>
          <tbody>
            {facts.map((f) => (
              <tr key={f.callId} className={TR}>
                <td className={TD}>
                  <Link href={`/calls/${f.callId}`} className={TD_LINK}>
                    {names.get(f.customerId) ?? f.customerId}
                  </Link>
                </td>
                <td className={TD_META}>{shortDate(f.endedAt)}</td>
                <td className={TD_NUM}>{f.claims}</td>
                <td className={TD_NUM_MUTED}>{f.verified}</td>
                <td className={TD_NUM}>{f.wrong}</td>
                <td className={TD_NUM_MUTED}>{f.stale}</td>
                <td className={TD_NUM_MUTED}>{f.unsupported}</td>
                <td className={f.coverageComplete ? TD_NUM_MUTED : TD_NUM}>
                  {f.coverageChecked}/{f.coverageUtterances}
                </td>
                <td className={TD_NUM_MUTED}>{rupees(f.costInr)}</td>
              </tr>
            ))}
          </tbody>
        </table>
      </section>
    </Shell>
  );
}
