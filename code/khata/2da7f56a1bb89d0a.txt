/**
 * Inbound call flow: the state a call moves through before anyone knows
 * who is on the line.
 *
 * The console used to open with a customer already selected, which
 * quietly assumed the rep knew who was ringing. On a real inbound call
 * they do not. They answer, hear a name, and look it up while talking.
 *
 * Pure functions with no imports, so lib/selfcheck.ts can assert the
 * search and the ticket guard directly.
 */

export type CallPhase =
  /** Nothing happening. The rep has not armed the line. */
  | "idle"
  /** Armed and ringing. No customer, no consent, nothing captured. */
  | "incoming"
  /** Answered. Consent may be taken; the caller is not yet identified. */
  | "answered"
  /** A customer record is loaded, or the rep marked the caller unknown. */
  | "identified";

/** The subset of a customer this module needs. */
export interface SearchableCustomer {
  id: string;
  name: string;
  city: string;
}

/**
 * Customer search.
 *
 * IMPORTANT, AND THE REASON THIS IS NOT A DATABASE QUERY: the list handed
 * to this function is whatever /api/bootstrap returned, and bootstrap
 * already scoped it through listCustomersForUser — a WHERE clause on
 * assigned_user_id. A rep's browser never holds another rep's customers,
 * so search cannot surface one however it is spelled. Adding a search
 * endpoint here would mean a second scoping path to keep correct, and
 * two scoping paths is one too many.
 */
export function searchCustomers<T extends SearchableCustomer>(
  scoped: T[],
  query: string,
  limit = 8
): T[] {
  const q = query.trim().toLowerCase();
  if (q === "") return [];

  const scored: Array<{ row: T; rank: number }> = [];
  for (const row of scoped) {
    const name = row.name.toLowerCase();
    const city = row.city.toLowerCase();

    // Name beats city, prefix beats substring. A rep hearing "Meena…"
    // should not have to wait past a city called Meerut.
    let rank = -1;
    if (name.startsWith(q)) rank = 0;
    else if (name.split(/\s+/).some((part) => part.startsWith(q))) rank = 1;
    else if (name.includes(q)) rank = 2;
    else if (city.startsWith(q)) rank = 3;
    else if (city.includes(q)) rank = 4;

    if (rank >= 0) scored.push({ row, rank });
  }

  return scored
    .sort((a, b) => a.rank - b.rank || (a.row.name < b.row.name ? -1 : 1))
    .slice(0, limit)
    .map((s) => s.row);
}

/**
 * The ticket amount to adjudicate against, given who is on the call.
 *
 * Null for an unidentified caller. adjudicate() produces no probes for
 * emi_amount or total_payable without a positive ticket, so those claims
 * come back UNSUPPORTED — which is the honest verdict. We do not know the
 * basket, so we cannot know whether "your EMI is about nine and a half
 * thousand" is right, and saying VERIFIED or WRONG would both be lies.
 *
 * Everything that depends only on the product terms — fees, interest,
 * tenure, late fee, eligibility, KYC — is unaffected. Those are true or
 * false regardless of who is calling.
 */
export function ticketForCall(
  customer: { cart: { ticketInr: number } } | null
): number | null {
  const ticket = customer?.cart.ticketInr;
  if (typeof ticket !== "number" || !Number.isFinite(ticket) || ticket <= 0) {
    return null;
  }
  return ticket;
}

/** Claim types that cannot be checked without knowing the basket size. */
export const TICKET_DEPENDENT_CLAIMS = ["emi_amount", "total_payable"] as const;

/**
 * Why the khata cannot be written yet, or null when it can.
 *
 * A khata is filed against a customer row and the calls table has a
 * foreign key onto it, so an unidentified call has nowhere to file. The
 * call still runs and still adjudicates; it just cannot be closed until
 * the rep has named the caller.
 */
export function closeBlockedBecause(
  phase: CallPhase,
  hasCustomer: boolean
): string | null {
  if (phase === "idle") return "No call is in progress.";
  if (phase === "incoming") return "The call has not been answered yet.";
  if (!hasCustomer) {
    return "Identify the caller before closing. A khata is filed against a customer record, and this call is not attached to one yet. Search for them above, or the call ends without a receipt.";
  }
  return null;
}
