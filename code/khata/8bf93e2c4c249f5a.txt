/**
 * Whether the login screen prints the credentials.
 *
 * Default ON. Being locked out of your own demo is a worse failure than
 * a development build showing a development password, so this has to be
 * switched off deliberately rather than switched on deliberately.
 */
export function showDemoCredentials(): boolean {
  return process.env.NEXT_PUBLIC_KHATA_HIDE_DEMO_CREDENTIALS !== "1";
}
