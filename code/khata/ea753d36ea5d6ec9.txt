/**
 * GET /api/bootstrap — customers and deals for the console.
 *
 * The console used to import data/customers.json and data/crm.json at
 * module scope. SQLite cannot be read from a client component, so the
 * same data arrives over this route instead, in the shape the console
 * already expected.
 *
 * This file reads the cookie and serialises. What a role may see is
 * decided by buildBootstrap in lib/bootstrap.ts, which selfcheck calls
 * directly — the rule is one code path, not one written twice.
 *
 * data/product_truth.json is NOT here. It stays a static import in both
 * the console and the adjudicator, because it is a spec rather than a
 * dataset and both must read the identical bytes.
 */

import { NextResponse } from "next/server";

import { authDisabled } from "@/lib/auth";
import { buildBootstrap } from "@/lib/bootstrap";
import { currentSession } from "@/lib/session";

export const runtime = "nodejs";
export const dynamic = "force-dynamic";

export type {
  BootstrapCustomer,
  BootstrapDeal,
  BootstrapResponse,
  BootstrapUser,
} from "@/lib/bootstrap";

export async function GET() {
  const session = await currentSession();
  return NextResponse.json(buildBootstrap(session, authDisabled()));
}
